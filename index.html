/** * ACHI - Integrated Management System 2026 * Developer: Gemini for Boss */ function doGet() { return HtmlService.createHtmlOutputFromFile('index') .setTitle("ACHI System") .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } // Fungsi untuk memanggil file HTML lain ke dalam index function include(filename) { return HtmlService.createHtmlOutputFromFile(filename).getContent(); } function doPost(e) { var action = e.parameter.action; var payload = JSON.parse(e.postData.contents); if (action === 'login') { return ContentService.createTextOutput(JSON.stringify(loginProcess(payload))) .setMimeType(ContentService.MimeType.JSON); } } function loginProcess(data) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var userSheet = ss.getSheetByName("Users"); // Deteksi sheet transaksi secara otomatis (cari Data_Log atau sheet pertama) var transSheet = ss.getSheetByName("Data_Log") || ss.getSheets()[0]; if (!userSheet) return { success: false, message: "Database 'Users' tidak ditemukan!" }; var userValues = userSheet.getDataRange().getValues(); for (var i = 1; i < userValues.length; i++) { var dbUser = userValues[i][0].toString().trim(); var dbPass = userValues[i][1].toString().trim(); if (dbUser.toLowerCase() === data.username.toLowerCase() && dbPass === data.password.toString()) { var role = userValues[i][2]; var u = dbUser.toUpperCase(); // Identifikasi Owner Spesifik var isOwner = (role === "Owner" || u === "HENGKY" || u === "ACHI" || u === "CHRISTY NOVALITA PRATIWI"); // Data statistik khusus Owner & Accounting var stats = { pemasukan: 0, pengeluaran: 0 }; if (isOwner || role === "Accounting") { stats = calculateTripStats(transSheet); } return { success: true, user: { username: dbUser, role: isOwner ? "Owner" : role, kapal: userValues[i][3], foto: userValues[i][4] || "", saldo: userValues[i][5] || 0, hutang: userValues[i][6] || 0 }, stats: stats }; } } return { success: false, message: "Username atau Password salah!" }; } catch (err) { return { success: false, message: "Sistem Error: " + err.message }; } } function calculateTripStats(sheet) { if (!sheet) return { pemasukan: 0, pengeluaran: 0 }; var data = sheet.getDataRange().getValues(); var pmsk = 0; var pglr = 0; for (var i = 1; i < data.length; i++) { var nilai = parseFloat(data[i][9]) || 0; // Kolom J (Harga Total) var kategori = data[i][5].toString().toLowerCase(); // Kolom F (Keterangan) if (kategori.includes("jual") || kategori.includes("masuk") || kategori.includes("setoran")) { pmsk += nilai; } else { pglr += nilai; } } return { pemasukan: pmsk, pengeluaran: pglr }; }
